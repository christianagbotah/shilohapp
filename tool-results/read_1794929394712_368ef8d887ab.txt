    69→    69→export function useAdminOrders() {
    70→    70→  const token = useUi((s) => s.adminToken);
    71→    71→  const setAdminToken = useUi((s) => s.setAdminToken);
    72→    72→  return useQuery({
    73→    73→    queryKey: ["admin-orders", token],
    74→    74→    refetchInterval: 30_000,
    75→    75→    queryFn: async () => {
    76→    76→      try {
    77→    77→        return await fetchJson<AdminOrdersResponse>("/api/admin/orders", {
    78→    78→          headers: { "x-admin-token": token ?? "" },
    79→    79→        });

    86→    86→    },

   432→   432→function exportOrdersCsv(orders: OrderDTO[]) {
   433→   433→  const header = [
   434→   434→    "Code",
   435→   435→    "Placed at",
   436→   436→    "Customer",
   437→   437→    "Phone",
   438→   438→    "Address",
   439→   439→    "Items",
   440→   440→    "Total (GHS)",
   441→   441→    "Payment status",
   442→   442→    "Payment method",
   443→   443→    "Paid at",
   444→   444→    "Order status",
   445→   445→    "Notes",
   446→   446→  ];
   447→   447→  const escape = (cell: string) => `"${cell.replace(/"/g, '""')}"`;
   448→   448→  const rows = orders.map((o) => [
   449→   449→    o.code,
   450→   450→    new Date(o.createdAt).toLocaleString("en-GB"),
   451→   451→    o.fullName,
   452→   452→    o.phone,
   453→   453→    o.address ?? "",
   454→   454→    o.items
   455→   455→      .map((i) => `${i.quantity}x ${i.productName} (${i.color}/${i.size}${i.capStyle ? `/${i.capStyle}` : ""})`)
   456→   456→      .join("; "),
   457→   457→    o.total.toFixed(2),
   458→   458→    o.paymentStatus,
   459→   459→    o.paymentMethod ?? "",
   460→   460→    o.paidAt ? new Date(o.paidAt).toLocaleString("en-GB") : "",
   461→   461→    o.status,
   462→   462→    o.adminNotes ?? "",
   463→   463→  ].map(escape).join(","));
   464→   464→  const csv = [header.map(escape).join(","), ...rows].join("\r\n");
   465→   465→  const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8" });
   466→   466→  const url = URL.createObjectURL(blob);
   467→   467→  const a = document.createElement("a");
   468→   468→  a.href = url;
   469→   469→  a.download = `shiloh-orders-${new Date().toISOString().slice(0, 10)}.csv`;
   470→   470→  document.body.appendChild(a);
   471→   471→  a.click();
   472→   472→  a.remove();
   473→   473→  URL.revokeObjectURL(url);
   474→   474→}
