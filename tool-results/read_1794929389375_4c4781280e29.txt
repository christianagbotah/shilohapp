    69→export function useAdminOrders() {
    70→  const token = useUi((s) => s.adminToken);
    71→  const setAdminToken = useUi((s) => s.setAdminToken);
    72→  return useQuery({
    73→    queryKey: ["admin-orders", token],
    74→    refetchInterval: 30_000,
    75→    queryFn: async () => {
    76→      try {
    77→        return await fetchJson<AdminOrdersResponse>("/api/admin/orders", {
    78→          headers: { "x-admin-token": token ?? "" },
    79→        });
    80→      } catch (err) {
    81→        if (err instanceof Error && err.message === "Unauthorized") {
    82→          setAdminToken(null);
    83→        }
    84→        throw err;
    85→      }
    86→    },
    87→    enabled: !!token,
    88→  });
    89→}

   432→function exportOrdersCsv(orders: OrderDTO[]) {
   433→  const header = [
   434→    "Code",
   435→    "Placed at",
   436→    "Customer",
   437→    "Phone",
   438→    "Address",
   439→    "Items",
   440→    "Total (GHS)",
   441→    "Payment status",
   442→    "Payment method",
   443→    "Paid at",
   444→    "Order status",
   445→    "Notes",
   446→  ];
   447→  const escape = (cell: string) => `"${cell.replace(/"/g, '""')}"`;
   448→  const rows = orders.map((o) => [
   449→    o.code,
   450→    new Date(o.createdAt).toLocaleString("en-GB"),
   451→    o.fullName,
   452→    o.phone,
   453→    o.address ?? "",
   454→    o.items
   455→      .map((i) => `${i.quantity}x ${i.productName} (${i.color}/${i.size}${i.capStyle ? `/${i.capStyle}` : ""})`)
   456→      .join("; "),
   457→    o.total.toFixed(2),
   458→    o.paymentStatus,
   459→    o.paymentMethod ?? "",
   460→    o.paidAt ? new Date(o.paidAt).toLocaleString("en-GB") : "",
   461→    o.status,
   462→    o.adminNotes ?? "",
   463→  ].map(escape).join(","));
   464→  const csv = [header.map(escape).join(","), ...rows].join("\r\n");
   465→  const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8" });
   466→  const url = URL.createObjectURL(blob);
   467→  const a = document.createElement("a");
   468→  a.href = url;
   469→  a.download = `shiloh-orders-${new Date().toISOString().slice(0, 10)}.csv`;
   470→  document.body.appendChild(a);
   471→  a.click();
   472→  a.remove();
   473→  URL.revokeObjectURL(url);
   474→}
